/** How many batches an export of `rows` rows needs. */
export function batchCount(rows: number, batchSize = 500): number {
  if (!Number.isInteger(rows) || rows < 0) {
    throw new RangeError(`rows must be a whole number ≥ 0, got ${rows}`);
  }
  return Math.ceil(rows / batchSize);
}
// BatchCount returns how many batches an export of rows rows needs.
func BatchCount(rows, batchSize int) (int, error) {
	if rows < 0 {
		return 0, fmt.Errorf("rows must be ≥ 0, got %d", rows)
	}
	return (rows + batchSize - 1) / batchSize, nil
}
